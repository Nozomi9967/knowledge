## 题解

```java
class Solution {

    boolean isNebor(String a,String b){
        int n=a.length();
        int err=0;
        for(int i=0;i<n;i++){
            if(a.charAt(i)!=b.charAt(i)) err++;
            if(err>1) return false;
        }
        return true;
    }

    public int ladderLength(String beginWord, String endWord, List<String> wordList) {
        Set<String> set=new HashSet<>();
        wordList.forEach(s->set.add(s));
        if(!set.contains(endWord)) return 0;

        Queue<String> q=new LinkedList<>();
        q.offer(beginWord);
        Set<String> visited=new HashSet<>();
        visited.add(beginWord);

        int count=1;
        while(!q.isEmpty()){
            count++;
            int size=q.size();
            for(int i=0;i<size;i++){
                String start=q.poll();
                System.out.println(start);
                // 遍历wordList，添加与start相邻的单词
                wordList.forEach(word->{
                    if(!visited.contains(word)&&isNebor(word,start)){
                        visited.add(word);
                        q.offer(word);
                    }
                });
            }
            if(visited.contains(endWord)) break;
        }
        return visited.contains(endWord)?count:0;
    }
}
```
