https://leetcode.cn/problems/kth-smallest-element-in-a-sorted-matrix/description/


**整个矩阵不是严格递增的**
```java
class Solution {

    public int kthSmallest(int[][] matrix, int k) {

        int n=matrix.length;

        int l=matrix[0][0],r=matrix[n-1][n-1];

        while(l<=r){

            int m=(r-l)/2+l;

            if(check(m,matrix,k)) r=m-1;

            else l=m+1;

        }

        return l;

    }

  

    public boolean check(int m,int[][] matrix,int k){

        int cnt=0;

        int n=matrix.length;

        for(int i=0;i<n;i++){

            if(m>=matrix[i][n-1]){

                cnt+=n;

                continue;

            }

            for(int j=0;j<n;j++){

                if(m>=matrix[i][j]) cnt++;

            }

        }

        return cnt>=k;

    }

}
```
