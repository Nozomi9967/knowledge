这题有点阴，这个假如num=2 from=1 to=3的话，实际这俩人在3就已经下车了，所以实际闭区间应该是from , to-1，所以在加减差分数组时才是`d[to]-=num`
```java
class Solution {

    public boolean carPooling(int[][] trips, int capacity) {

        int[] d=new int[1001];

        int far=0;

        for(int[] t:trips){

            int num=t[0],from=t[1],to=t[2];

            d[from]+=num;

            d[to]-=num;

            far=Math.max(far,to);

        }

        int cur=0;

        for(int i=0;i<=far;i++){

            cur+=d[i];

            if(cur>capacity) return false;

        }

        return true;

  

    }

}

  

// 0,0 1,2 2,0 3,0 4,0 5,3 6,-2 7,0 8,-3
```
