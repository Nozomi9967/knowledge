## ==我的题解==

```java

class Solution {

    public int[] answerQueries(int[] nums, int[] queries) {

        Arrays.sort(nums);

        int n=nums.length;

        int m=queries.length;

        int[] s=new int[n+1];

        for(int i=1;i<=n;i++){

            s[i]=s[i-1]+nums[i-1];

        }

        int[] ans=new int[m];

        for(int i=0;i<m;i++){

            ans[i]=func(queries[i],s);

        }

        return ans;

    }

  

    public int func(int x,int[] s){

        int l=0,r=s.length-1;

        // System.out.println("x: "+x);

        while(l<=r){

            int m=(r-l)/2+l;

            // System.out.println("l: "+l+" m: "+m+" r: "+r+" s[m]: "+s[m]);

            if(s[m]<=x) l=m+1;

            else r=m-1;

        }

        return r;

    }

}

class Solution:
    def answerQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        nums.sort()
        for i in range(1, len(nums)):
            nums[i] += nums[i - 1]  # 原地求前缀和
        for i, q in enumerate(queries):
            queries[i] = bisect_right(nums, q)  # 复用 queries 作为答案
        return queries
```

class Solution {
public:
    // 左闭右开
    int lower_bound(vector<int>& nums, int target) {
        int left = 0, right = nums.size();
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] < target) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        return left;
    }

    vector<int> answerQueries(vector<int>& nums, vector<int>& queries) {
        ranges::sort(nums);
        int n = nums.size();
        // 求前缀和
        vector<int> pre_sum(n + 1, 0);
        for (int i = 0; i < nums.size(); i++) {
            pre_sum[i + 1] = pre_sum[i] + nums[i];
        }

        for (auto& x : queries) {
            // pre_sum中有先导0，所以要求出 >x 的第一个元素下标并-1
            x = lower_bound(pre_sum, x + 1) - 1;
        }
        return queries;
    }
};
```

class Solution:
    def answerQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        
        m = len(queries)
        ans = [0] * m
        nums.sort()
        tlt = 0

        new_queries = sorted([(q, i) for i, q in enumerate(queries)])

        idx = 0
        for query, i in new_queries:
            
            while idx < len(nums) and \
                tlt + nums[idx] <= query:
                tlt += nums[idx]
                idx += 1

            ans[i] = idx

        return ans
```
