import java.util.*;

public class Main {
    
    static int precedence(char op) {
        if (op == '+' || op == '-') return 1;
        if (op == '*' || op == '/') return 2;
        return 0;
    }
    
    static boolean isOpenBracket(char c) {
        return c == '(' || c == '[' || c == '{';
    }
    
    static boolean isCloseBracket(char c) {
        return c == ')' || c == ']' || c == '}';
    }
    
    static void applyOp(Deque<Integer> nums, Deque<Character> ops) {
        int b = nums.pop();
        int a = nums.pop();
        char op = ops.pop();
        switch (op) {
            case '+': nums.push(a + b); break;
            case '-': nums.push(a - b); break;
            case '*': nums.push(a * b); break;
            case '/': nums.push(a / b); break;
        }
    }
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next().trim();
        
        Deque<Integer> nums = new ArrayDeque<>();
        Deque<Character> ops = new ArrayDeque<>();
        
        int i = 0;
        while (i < s.length()) {
            char c = s.charAt(i);
            
            if (Character.isDigit(c)) {
                int num = 0;
                while (i < s.length() && Character.isDigit(s.charAt(i))) {
                    num = num * 10 + (s.charAt(i) - '0');
                    i++;
                }
                nums.push(num);
                continue;
            }
            
            if (isOpenBracket(c)) {
                ops.push(c);
            } else if (isCloseBracket(c)) {
                while (!ops.isEmpty() && !isOpenBracket(ops.peek())) {
                    applyOp(nums, ops);
                }
                ops.pop(); // 弹出对应的开括号
            } else if (c == '+' || c == '-' || c == '*' || c == '/') {
                // 处理一元负号（如表达式开头或括号后的负号）
                // 题目保证合法，一般不需要，但保留结构清晰
                while (!ops.isEmpty()
                        && !isOpenBracket(ops.peek())
                        && precedence(ops.peek()) >= precedence(c)) {
                    applyOp(nums, ops);
                }
                ops.push(c);
            }
            
            i++;
        }
        
        while (!ops.isEmpty()) {
            applyOp(nums, ops);
        }
        
        System.out.println(nums.peek());
    }
}
