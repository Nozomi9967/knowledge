## 题解



```java
class Solution {
    boolean[][] visited;
    int m, n;
    
    boolean dfs(char[][] board, String word, int index, int a, int b) {
        // 1. 先检查边界和访问状态
        if (a < 0 || a >= m || b < 0 || b >= n) return false;
        if (visited[a][b]) return false;
        
        // 2. 检查当前字符是否匹配
        if (board[a][b] != word.charAt(index)) return false;
        
        // 3. 检查是否找到完整单词
        if (index == word.length() - 1) return true;
        
        // 4. 标记当前格子为已访问
        visited[a][b] = true;
        
        // 5. 向四个方向探索
        boolean found = dfs(board, word, index + 1, a + 1, b) ||
                       dfs(board, word, index + 1, a - 1, b) ||
                       dfs(board, word, index + 1, a, b + 1) ||
                       dfs(board, word, index + 1, a, b - 1);
        
        // 6. 回溯：恢复标记
        visited[a][b] = false;
        
        return found;
    }
    
    public boolean exist(char[][] board, String word) {
        m = board.length;
        n = board[0].length;
        visited = new boolean[m][n];
        
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (dfs(board, word, 0, i, j)) {
                    return true;
                }
            }
        }
        return false;
    }
}
```

```java
class Solution {

    boolean[][] visited;

    boolean dfs(char[][] board, String word,int m,int n,int index,int a,int b){
        if(index==word.length()) return true;
        if(board[a][b]!=word.charAt(index)||visited[a][b]) return false;
        boolean b1=false,b2=false,b3=false,b4=false;
        if(a>=0&&b>=0&&a+1<m&&b<n){
            visited[a+1][b]=true;
            b1=dfs(board,word,m,n,index+1,a+1,b);
            visited[a+1][b]=b1;
        }
        if(a>=0&&b>=0&&a<m&&b+1<n){
            visited[a][b+1]=true;
            b2=dfs(board,word,m,n,index+1,a,b+1);
            visited[a][b+1]=b2;
        }
        if(a-1>=0&&b>=0&&a<m&&b<n){
            visited[a-1][b]=true;
            b3=dfs(board,word,m,n,index+1,a-1,b);
            visited[a-1][b]=b3;
        }
        if(a>=0&&b-1>=0&&a<m&&b<n){
            visited[a][b-1]=true;
            b4=dfs(board,word,m,n,index+1,a,b-1);
            visited[a][b-1]=b4;
        }
        return b1||b2||b3||b4;
    }

    public boolean exist(char[][] board, String word) {
        int m=board.length;
        int n=board[0].length;
        visited=new boolean[m][n];
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(dfs(board,word,m,n,0,i,j)) return true;
            }
        }
        return false;
    }
}
```
