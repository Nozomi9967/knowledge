```java
// 原错误题解,超出时间限制
class Solution {
    public int coinChange(int[] coins, int amount) {
        // dp[i]表示金额i需要最少dp[i]种硬币
        int[] dp=new int[amount+1];
        Arrays.fill(dp,-1);
        dp[0]=0;
        Set<Integer> set = new HashSet<>(Arrays.stream(coins).boxed().toList());
        for(int i=1;i<amount+1;i++){
            for(int j=0;j<i;j++){
                if(dp[j]!=-1&&set.contains(i-j)){
                    if(dp[i]!=-1) dp[i]=Math.min(dp[j]+1,dp[i]);
                    else dp[i]=dp[j]+1;
                }
            }
        }
        return dp[amount];
    }
}
```

```java
class Solution {
    public int coinChange(int[] coins, int amount) {
        // dp[i]表示金额i需要最少dp[i]种硬币
        int[] dp=new int[amount+1];
        Arrays.fill(dp,-1);
        dp[0]=0;
        Set<Integer> set = new HashSet<>(Arrays.stream(coins).boxed().toList());
        // 遍历不同面额的硬币更快
        for(int coin:coins){
            for(int i=1;i<=amount;i++){
                if(i-coin>=0&&dp[i-coin]!=-1){
                    if(dp[i]!=-1) dp[i]=Math.min(dp[i],dp[i-coin]+1);
                    else dp[i]=dp[i-coin]+1;
                }
            }
        }
        return dp[amount];
    }
```
