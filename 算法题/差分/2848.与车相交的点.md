https://leetcode.cn/problems/points-that-intersect-with-cars/

```java
class Solution {

    public int numberOfPoints(List<List<Integer>> nums) {

        int[] d=new int[102];

        nums.forEach(

            num->{

                int start=num.get(0),end=num.get(1);

                d[start]++;

                d[end+1]--;

            }

        );

        int s=0;

        int ans=0;

        for(int i=1;i<=100;i++){

            s+=d[i];

            if(s>0) ans++;

        }

        return ans;

    }

}
```

```java
class Solution {

    public int numberOfPoints(List<List<Integer>> nums) {

        int[] d=new int[102];

        for(List<Integer> t:nums){

            d[t.get(0)]++;

            d[t.get(1)+1]--;

        }

        int ans=0;

        for(int i=1,cur=0;i<=100;i++){

            cur+=d[i];

            if(cur>=1) ans++;

        }

        return ans;

    }

}
```
