class Solution {

    public int[][] matrixBlockSum(int[][] mat, int k) {

        int m=mat.length;

        int n=mat[0].length;

        int[][] s=new int[m][n];

        for(int i=0;i<m;i++){

            for(int j=0;j<n;j++){

                int x= i-1>=0?s[i-1][j]:0;

                int y= j-1>=0?s[i][j-1]:0;

                int z= i-1>=0&&j-1>=0?s[i-1][j-1]:0;

                s[i][j]=x+y-z+mat[i][j];

            }

        }

  

        int[][] ans=new int[m][n];

        for(int i=0;i<m;i++){

            for(int j=0;j<n;j++){

                // 求max和min的xy

                // max和min都是代表点

                // 理论上max(i+k,j+k) min(i-k,j-k)

                // 但由于可能越界 所以需要额外判断

                // 并且由于我没有在前缀和数组外加一圈 所以需要对minX minY为0的情况额外判断

                int maxX=i+k<m?i+k:m-1;

                int maxY=j+k<n?j+k:n-1;

                int minX=i-k-1>=0?i-k-1:-1;

                int minY=j-k-1>=0?j-k-1:-1;

                int sum=s[maxX][maxY];

                int x=minY==-1?0:s[maxX][minY];

                int y=minX==-1?0:s[minX][maxY];

                int z=minX==-1||minY==-1?0:s[minX][minY];

                ans[i][j]=sum-x-y+z;

            }

        }

        return ans;

    }

}
