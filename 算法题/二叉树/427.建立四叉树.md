class Solution {

    Node dfs(int[][] grid,int r,int c,int n){
        int r0=r;
        int c0=c;
        int r1=r+n;
        int c1=c+n;
        for(int i=r0;i<r1;i++){
            for(int j=c0;j<c1;j++){
                if(grid[i][j]!=grid[r0][c0]){
                    return new Node(
                        true,
                        false,
                        dfs(grid,r0,c0,n/2),
                        dfs(grid,r0,c0+n/2,n/2),
                        dfs(grid,r0+n/2,c0,n/2),
                        dfs(grid,r0+n/2,c0+n/2,n/2)
                    );
                }
            }
        }
        return new Node(1==grid[r0][c0],true,null,null,null,null);
    }

    public Node construct(int[][] grid) {
        return dfs(grid,0,0,grid.length);
    }
}
