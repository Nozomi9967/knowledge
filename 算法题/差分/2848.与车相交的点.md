class Solution {

    public int numberOfPoints(List<List<Integer>> nums) {

        int[] d=new int[102];

        nums.forEach(

            num->{

                int start=num.get(0),end=num.get(1);

                d[start]++;

                d[end+1]--;

            }

        );

        int s=0;

        int ans=0;

        for(int i=1;i<=100;i++){

            s+=d[i];

            if(s>0) ans++;

        }

        return ans;

    }

}
