## 题目

https://leetcode.cn/problems/course-schedule/description/?envType=study-plan-v2&envId=top-interview-150





## 题解

有向图、拓扑图



```java
class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        List<List<Integer>> adjList=new ArrayList<>();// 表示有向边的数组
        int[] inDegree=new int[numCourses];// 表示入度的数组
        for(int i=0;i<numCourses;i++) adjList.add(new ArrayList<Integer>()); // 初始化
        // 构造有向边、入度
        for(int[] edge:prerequisites){
            int start=edge[0];
            int end=edge[1];
            adjList.get(start).add(end);
            inDegree[end]++;
        }
		
        // 队列，入度为0则入队
        Queue<Integer> q=new LinkedList<>();
        for(int i=0;i<numCourses;i++){
            if(inDegree[i]==0) q.offer(i);
        }
        int count=0;
        // 对每一个入度为0的点的出边数组进行遍历，该点所指向的点的入度-1
        // -1后如果入度为0，则入队，重复直到队列中为空
        while(!q.isEmpty()){
            count++;
            int start=q.poll();
            List<Integer> list=adjList.get(start);
            for(int i=0;i<list.size();i++){
                int end=list.get(i);
                inDegree[end]--;
                if(inDegree[end]==0) q.offer(end);
            }
        }
        // 如果最后入度为0的点数 == 总点数  则没有环，可以完成学习
        return count==numCourses;
    }
}
```
