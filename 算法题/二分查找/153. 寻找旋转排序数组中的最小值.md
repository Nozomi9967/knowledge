```java
class Solution {
    public int findMin(int[] nums) {
        int left=0;
        int right=nums.length;
        int last=nums[right-1];
        while(left!=right){
            int mid=left+(right-left)/2;
            if(nums[mid]>last) left=mid+1;
            else right=mid;
        }
        return nums[right];
    }
}
```
