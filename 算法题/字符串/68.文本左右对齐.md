class Solution {
    public List<String> fullJustify(String[] words, int maxWidth) {
        int wordNum=0;
        int charNum=0;
        int index=0;
        List<String> list=new ArrayList<>();
        while(true){
            int tem=(wordNum==0)?0:wordNum;
            if(index<words.length&&charNum+words[index].length()+tem<=maxWidth){
                wordNum++;
                charNum+=words[index].length();
                index++;
                continue;
            }else{
                int spaceNum=maxWidth-charNum;
                int avrSpace=(wordNum==1)?spaceNum:spaceNum/(wordNum-1);
                int flag=(wordNum==1)?0:spaceNum%(wordNum-1);
                StringBuilder sb=new StringBuilder("");
                for(int i=index-wordNum;i<index;i++){
                    
                    if(index!=words.length){
                        if(i!=index-wordNum) {
                        int repeatNum=(flag-->0)?avrSpace+1:avrSpace;
                        spaceNum-=repeatNum;
                        sb.append(" ".repeat(repeatNum));
                        }
                        
                        sb.append(words[i]);

                        if(wordNum==1) sb.append(" ".repeat(spaceNum));
                    }else{
                            sb.append(words[i]);
                        if(i==index-1){
                            sb.append(" ".repeat(spaceNum>0?spaceNum:0));
                        }else{
                            sb.append(" ");
                            spaceNum--;
                        }
                    }
                }
                list.add(sb.toString());
                charNum=0;
                wordNum=0;
                if(index==words.length) break;
            }
        }
        return list;
        
    }
}
