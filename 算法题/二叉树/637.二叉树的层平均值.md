## 题目

https://leetcode.cn/problems/average-of-levels-in-binary-tree/description/?envType=study-plan-v2&envId=top-interview-150





## 题解

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Double> averageOfLevels(TreeNode root) {
        List<Double> res=new ArrayList<>();
        res.add(Double.valueOf(root.val));
        List<TreeNode> floor=List.of(root);
        while(!floor.isEmpty()){
            List<TreeNode> ceil=floor;
            Double sum=0.0;
            floor=new ArrayList<>();
            for(int i=0;i<ceil.size();i++){
                TreeNode n=ceil.get(i);
                if(n.left!=null){
                     floor.add(n.left);
                     sum+=n.left.val;
                }
                if(n.right!=null){
                     floor.add(n.right);
                     sum+=n.right.val;
                }
            }
            if(floor.size()!=0) res.add(Double.valueOf(sum/floor.size()));
        }
        return res;
    }
}
```
