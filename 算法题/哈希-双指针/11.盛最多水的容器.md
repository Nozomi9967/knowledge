// 我的题解--双指针
// 能盛多少水取决于最短的那个
class Solution {
    public int maxArea(int[] height) {
        int left=0,right=height.length-1;
        int max=0;
        while(left<right){
            int num=(right-left)*Math.min(height[left],height[right]);
            max=num>max?num:max;
            if(height[left]>=height[right]) right--;
            else left++;
        }
        return max;
    }
}
