// 我的
// 将所有的String都和第一个String的0到len-1个char相对比
// 一样则添加该char，否则直接返回
class Solution {
    public String longestCommonPrefix(String[] strs) {
        StringBuilder maxPrefix=new StringBuilder("");
        int i=0;
        while(true){
            if(i>=strs[0].length()) break;
            for(int j=1;j<strs.length;j++){
                if(i>=strs[j].length()) return maxPrefix.toString();
                if(strs[j].charAt(i)!=strs[0].charAt(i)) return maxPrefix.toString();
            }
            maxPrefix.append(strs[0].charAt(i));
            i++;
        }
        return maxPrefix.toString();
    }
}
