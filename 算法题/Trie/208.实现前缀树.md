```java
class Trie {

    TreeNode node;

    class TreeNode{
        Character c;
        Map<Character,TreeNode> map;
        boolean isEnd;

        public TreeNode(){}

        public TreeNode(Character c,Map<Character,TreeNode> map,boolean isEnd){
            this.c=c;
            this.map=map;
            this.isEnd=isEnd;
        }
    }

    public Trie() {
        node=new TreeNode();
        node.c='@';
        node.map=new HashMap<Character,TreeNode>();
        node.isEnd=false;
    }
    
    public void insert(String word) {
        Map<Character,TreeNode> m=node.map;
        int len=word.length();
        for(int i=0;i<len;i++){
            char ch=word.charAt(i);
            if(m.containsKey(ch)&&i==len-1){
                m.get(ch).isEnd=true;
                return;
            }
            if(m.containsKey(ch)) m=m.get(ch).map;
            else {
                TreeNode t=new TreeNode(ch,new HashMap<Character,TreeNode>(),i==len-1);
                System.out.println(ch);
                m.put(ch,t);
                m=t.map;
            }
        }
    }
    
    public boolean search(String word) {
        Map<Character,TreeNode> m=node.map;
        int len=word.length();
        for(int i=0;i<len;i++){
            char ch=word.charAt(i);
            // System.out.println(ch);
            if(!m.containsKey(ch)) return false;
            if(i==len-1&&m.get(ch).isEnd) return true;
            m=m.get(ch).map;
        }
        return false;
    }
    
    public boolean startsWith(String prefix) {
        Map<Character,TreeNode> m=node.map;
        for(int i=0;i<prefix.length();i++){
            char ch=prefix.charAt(i);
            if(!m.containsKey(ch)) return false;
            m=m.get(ch).map;
        }
        return true;
    }
}
```

```java
class Trie {
    private static class Node {
        Node[] son = new Node[26];
        boolean end = false;
    }

    private final Node root = new Node();

    public void insert(String word) {
        Node cur = root;
        for (char c : word.toCharArray()) {
            c -= 'a';
            if (cur.son[c] == null) { // 无路可走？
                cur.son[c] = new Node(); // new 出来！
            }
            cur = cur.son[c];
        }
        cur.end = true;
    }

    public boolean search(String word) {
        return find(word) == 2;
    }

    public boolean startsWith(String prefix) {
        return find(prefix) != 0;
    }

    private int find(String word) {
        Node cur = root;
        for (char c : word.toCharArray()) {
            c -= 'a';
            if (cur.son[c] == null) { // 道不同，不相为谋
                return 0;
            }
            cur = cur.son[c];
        }
        // 走过同样的路（2=完全匹配，1=前缀匹配）
        return cur.end ? 2 : 1;
    }
}
```
