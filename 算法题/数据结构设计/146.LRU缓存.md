### 题解

```java
class LRUCache {

    private Integer capacity;

    private Integer size;

    private Node head;
    private Node tail;

    private Map<Integer,Node> map=new HashMap<>();

    class Node{
        public Node pre;
        public Node next;
        public Integer key;
        public Integer value;
    }

    public LRUCache(int capacity) {
        this.head=new Node();
        this.tail=this.head;
        this.capacity=capacity;
        this.size=0;
    }
    
    public int get(int key) {
        Node hot=this.map.get(key);
        // 不存在
        if(hot==null) return -1;
        // 只有一个元素
        if(hot.pre==null&&hot.next==null) return hot.value;
        // get的元素在头部
        if(hot.pre==null) return this.head.value;
        // get的元素在尾部
        if(hot.next==null){
            this.tail=this.tail.pre;
        }
        hot.pre.next=hot.next;
        if(hot.next!=null) hot.next.pre=hot.pre;
        hot.pre=null;
        hot.next=this.head;
        this.head.pre=hot;
        this.head=hot;
        // printLink();
        return this.head.value;
    }

    private void printLink(){
        Node p=head;
        while(p!=null){
            System.out.println(p.key+":"+p.value);
            p=p.next;
        }
        System.out.println("end");
    }
    
    public void put(int key, int value) {

        if(map.containsKey(key)){
            // 修改node.value
            Node n=map.get(key);
            n.value=value;
            // 更新链表
            get(key);
            return ;
        }

        if(size+1<=capacity){
            if(size==0){
                this.head.key=key;
                this.head.value=value;
                this.map.put(key,head);
            }else{
                Node n=new Node();
                n.value=value;
                n.key=key;
                this.head.pre=n;
                n.next=this.head;
                this.head=n;
                // this.tail.next=n;
                // n.pre=this.tail;
                // this.tail=n;
                this.map.put(key,n);
            }
            size++;
        }else{
            Node n=new Node();
            n.value=value;
            n.key=key;
            this.head.pre=n;
            n.next=this.head;
            this.head=n;
            // printLink();
            this.tail=this.tail.pre;
            this.map.remove(this.tail.next.key);
            this.tail.next=null;
            // if(this.tail.pre!=null) this.tail.pre.next=n;
            this.map.put(key,n);
        }
        // printLink();
    }
}

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache obj = new LRUCache(capacity);
 * int param_1 = obj.get(key);
 * obj.put(key,value);
 */
```
