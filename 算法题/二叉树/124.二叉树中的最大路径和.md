## 题解



```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    private int max=-1001;

    private int getMaxSum(TreeNode root){
        if(root==null){
             return 0;
        }
        if(root.left==null&&root.right==null){
            max=(root.val>max)?root.val:max;
             return root.val;
        }
        int sum=root.val;
        int left=getMaxSum(root.left);
        int right=getMaxSum(root.right);
        sum+=((left>0)?left:0);
        sum+=((right>0)?right:0);
        // System.out.println(sum);
        max=(sum>max)?sum:max;
        return sum;
    }

    public int maxPathSum(TreeNode root) {
        getMaxSum(root);
        return max;
    }
}
```

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    private int max=-1001;

    private int getMaxSum(TreeNode root){
        if(root==null){
             return 0;
        }
        if(root.left==null&&root.right==null){
            max=(root.val>max)?root.val:max;
             return root.val;
        }
        int sum=root.val;
        int left=Math.max(0,getMaxSum(root.left));
        int right=Math.max(0,getMaxSum(root.right));
        sum+=(left+right);
        max=(sum>max)?sum:max;
        return root.val+Math.max(left,right);
        // 重点是理清楚最大贡献值和最大路径和的区别
    }

    public int maxPathSum(TreeNode root) {
        getMaxSum(root);
        return max;
    }
}
```
