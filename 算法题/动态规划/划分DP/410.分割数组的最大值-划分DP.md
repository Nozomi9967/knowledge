```java
class Solution {

    public int splitArray(int[] nums, int t) {

        final int INF=Integer.MAX_VALUE/2;

        int n=nums.length;

        // d[i][j]前i个元素分为j个子数组时的各自和最大值中的最小值

        int[][] d=new int[n+1][t+1];

        for(int[] row:d) Arrays.fill(row,INF);

        int[] prefix=new int[n+1];

        for(int i=1;i<=n;i++) prefix[i]=prefix[i-1]+nums[i-1];

        d[0][0]=0;

        for(int i=1;i<=n;i++){

            for(int j=1;j<=Math.min(i,t);j++){

                for(int k=j-1;k<i;k++){

                    if(d[k][j-1]==INF) continue;

                    int lastSum=prefix[i]-prefix[k];

                    d[i][j]=Math.min(Math.max(d[k][j-1],lastSum),d[i][j]);

                }

            }

        }

        return d[n][t];

    }

}
```
