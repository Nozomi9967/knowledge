## 题目

## 题解

解法参考92，在此基础上增加计数器并循环



```java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode reverseKGroup(ListNode head, int k) {
        if(head.next==null) return head;
        ListNode dummy=new ListNode(0,head);
        int len=0;
        while(head!=null){
            len++;
            head=head.next;
        }
        if(len<k) return dummy.next;

        ListNode pre=null;
        ListNode cur=dummy.next;
        ListNode next=null;
        ListNode start=dummy;
        while(true){
            for(int i=0;i<k;i++){
                next=cur.next;
                cur.next=pre;
                pre=cur;
                cur=next;
            }
            ListNode newStart=start.next;
            start.next.next=cur;
            start.next=pre;
            start=newStart;
            pre=null;
            // System.out.println(start.val);
            cur=start.next;
            len-=k;
            if(len<k) break;
        }
        return dummy.next;
    }
}
```

```java
class Solution {

    public ListNode reverseKGroup(ListNode head, int k) {

        int n=0;

        ListNode p=head;

        while(p!=null){

            n++;

            p=p.next;

        }

        // if(n<k) return head;

        p=head;

        int i=0;

        ListNode pre=null,cur=p,next=null,dummy=p;

        ListNode ans=head;

        while(i<n&&i+k<=n){

            for(int j=0;j<k;j++){

                next=cur.next;

                cur.next=pre;

                pre=cur;

                cur=next;

            }

  

            if(i==0) ans=pre;

            i+=k;

            if(dummy.next!=null) dummy.next.next=pre;

            dummy.next=next;

            pre=dummy;

            cur=dummy=next;

  

        }

        return ans;

    }

}
```
