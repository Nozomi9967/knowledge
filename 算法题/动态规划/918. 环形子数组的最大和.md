class Solution {
    public int maxSubarraySumCircular(int[] nums) {
        int n=nums.length;
        int cur=nums[0];
        int max=nums[0];
        int sum=nums[0];
        for(int i=1;i<n;i++){
            cur=Math.max(nums[i],cur+nums[i]);
            max=Math.max(max,cur);
            sum+=nums[i];
        }
        int min=nums[0];
        cur=nums[0];
        for(int i=1;i<n;i++){
            cur=Math.min(nums[i],nums[i]+cur);
            min=Math.min(cur,min);
        }
        return sum==min?max:Math.max(max,sum-min);
    }
}
