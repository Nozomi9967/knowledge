
## 题目

https://leetcode.cn/problems/number-of-sub-arrays-with-odd-sum/

```java
class Solution {

    public int numOfSubarrays(int[] arr) {

        int n=arr.length;

        int[] s=new int[n+1];

        for(int i=1;i<=n;i++){

            s[i]=s[i-1]+arr[i-1];

        }

  

        Map<Character,Integer> cnt=new HashMap<>();

        // s[j]-s[i] % 2 == 1

        int ans=0;

        for(int sj:s){

            if(sj%2==1){

                ans+=cnt.getOrDefault('O',0);

                cnt.merge('J',1,Integer::sum);

            }else{

                ans+=cnt.getOrDefault('J',0);

                cnt.merge('O',1,Integer::sum);

            }

            ans=(int)(ans%(Math.pow(10,9)+7));

        }

        return ans;

    }

}
```
