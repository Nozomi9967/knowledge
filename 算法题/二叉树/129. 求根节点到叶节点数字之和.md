## 题解



```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    int ans=0;

    void dfs(TreeNode root,int x){
        if(root==null) return;
        x=x*10+root.val; // 这里每次就是绕不过来弯
        if(root.left==null&&root.right==null) ans+=x; // 叶子节点时再+
        dfs(root.left,x);
        dfs(root.right,x);
    }

    public int sumNumbers(TreeNode root) {
        dfs(root,0);
        return ans;
    }
}
```
