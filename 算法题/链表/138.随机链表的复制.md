## 题解

```c++
class Solution {
public:
    Node *copyRandomList(Node *head) {        
        unordered_map<Node *, Node *> mp;
        for (auto p = head; p; p = p->next) {
            mp[p] = new Node(p->val);            
        }
        for (auto p = head; p; p = p->next) {
            mp[p]->next = mp[p->next];
            mp[p]->random = mp[p->random];
        }  
        return mp[head];
    }
};
```

```java
class Solution {
    public Node copyRandomList(Node head) {
        if(head==null) return null;
        Map<Node,Integer> map1=new HashMap<>();
        // Map<Interger,Node> map2=new HashMap<>();
        Map<Integer,Node> map3=new HashMap<>();
        Node p=head;
        Node res=new Node(-1);
        Node q=res;
        int i=0;
        while(p!=null){
            map1.put(p,i);
            map3.put(i,q);
            q.val=p.val;
            p=p.next;
            if(p==null) break;
            Node x=new Node(-1);
            q.next=x;
            q=x;
            i++;
        }
        p=head;
        q=res;
        while(p!=null){
            Integer randomIndex=map1.get(p.random);
            q.random=map3.get(randomIndex);
            p=p.next;
            q=q.next;
        }
        return res;
    }
}
```
