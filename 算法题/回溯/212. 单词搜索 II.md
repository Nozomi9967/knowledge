## 题解

**原版：**

```java
class Solution {

    boolean bfs(int i,int j,int index,String word,char[][] board,boolean[][] visited){
        int m=board.length;
        int n=board[0].length;
        if(i<0||j<0||i>=m||j>=n||board[i][j]!=word.charAt(index)||visited[i][j]) return false;
        if(board[i][j]==word.charAt(index)&&index==word.length()-1) return true;
        visited[i][j]=true;
        boolean[][] v1=new boolean[m][n];
        boolean[][] v2=new boolean[m][n];
        boolean[][] v3=new boolean[m][n];
        boolean[][] v4=new boolean[m][n];
        for(int k=0;k<m;k++){
            v1[k]=Arrays.copyOf(visited[k],n);
            v2[k]=Arrays.copyOf(visited[k],n);
            v3[k]=Arrays.copyOf(visited[k],n);
            v4[k]=Arrays.copyOf(visited[k],n);
        }
        return bfs(i-1,j,index+1,word,board,v1)||bfs(i+1,j,index+1,word,board,v2)
                    ||bfs(i,j-1,index+1,word,board,v3)||bfs(i,j+1,index+1,word,board,v4);
    }

    public List<String> findWords(char[][] board, String[] words) {
        int len=words.length;
        int m=board.length;
        int n=board[0].length;
        List<String> res=new ArrayList<>();
        for(String word:words){
            // 寻找开始字符，递归
            boolean flag=false;
            for(int i=0;i<m&&!flag;i++){
                for(int j=0;j<n&&!flag;j++){
                    if(word.charAt(0)==board[i][j]&&bfs(i,j,0,word,board,new boolean[m][n])){
                        flag=true;
                        res.add(word);
                    }
                }
            }
        }
        return res;
    }
}
```

**豆包优化后：**

```java
import java.util.*;

class Solution {
    // 定义前缀树节点
    static class TrieNode {
        String word; // 存储完整单词（仅在叶子节点/匹配完成时赋值）
        TrieNode[] children = new TrieNode[26]; // 26个小写字母
    }

    // 构建前缀树
    private TrieNode buildTrie(String[] words) {
        TrieNode root = new TrieNode();
        for (String word : words) {
            TrieNode node = root;
            for (char c : word.toCharArray()) {
                int idx = c - 'a';
                if (node.children[idx] == null) {
                    node.children[idx] = new TrieNode();
                }
                node = node.children[idx];
            }
            node.word = word; // 单词结束，存储完整单词
        }
        return root;
    }

    // DFS回溯搜索
    private void dfs(char[][] board, int i, int j, TrieNode node, List<String> res, boolean[][] visited) {
        int m = board.length;
        int n = board[0].length;
        // 边界/已访问/无匹配节点：剪枝
        if (i < 0 || j < 0 || i >= m || j >= n || visited[i][j] || node.children[board[i][j] - 'a'] == null) {
            return;
        }

        // 移动到匹配的子节点
        node = node.children[board[i][j] - 'a'];
        // 找到完整单词：加入结果，并清空node.word避免重复添加
        if (node.word != null) {
            res.add(node.word);
            node.word = null; // 关键：去重，避免同一单词多次加入
        }

        // 回溯核心：标记已访问
        visited[i][j] = true;
        // 上下左右递归
        dfs(board, i - 1, j, node, res, visited);
        dfs(board, i + 1, j, node, res, visited);
        dfs(board, i, j - 1, node, res, visited);
        dfs(board, i, j + 1, node, res, visited);
        // 回溯核心：恢复未访问 这里重点	
        visited[i][j] = false;
    }

    public List<String> findWords(char[][] board, String[] words) {
        List<String> res = new ArrayList<>();
        if (board == null || board.length == 0 || words == null || words.length == 0) {
            return res;
        }

        // 1. 构建前缀树
        TrieNode root = buildTrie(words);
        int m = board.length;
        int n = board[0].length;
        boolean[][] visited = new boolean[m][n]; // 全局复用，回溯修改

        // 2. 遍历棋盘每个起点，DFS搜索
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                dfs(board, i, j, root, res, visited);
            }
        }
        return res;
    }

    // 测试示例
    public static void main(String[] args) {
        Solution solution = new Solution();
        char[][] board = {
                {'o', 'a', 'a', 'n'},
                {'e', 't', 'a', 'e'},
                {'i', 'h', 'k', 'r'},
                {'i', 'f', 'l', 'v'}
        };
        String[] words = {"oath", "pea", "eat", "rain"};
        System.out.println(solution.findWords(board, words)); // 输出 [oath, eat]
    }
}
```
