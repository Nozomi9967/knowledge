## 题解

```java
class Solution {
    public int snakesAndLadders(int[][] board) {
        Set<Integer> points=new HashSet<>();
        int n=board.length;
        int n2=n*n;
        boolean flag=false;
        int num=1;
        Map<Integer,Integer> targetMap=new HashMap<>();
        for(int i=n-1;i>=0;i--){
            int j=flag?n-1:0;
            while(j>=0&&j<n){
                if(board[i][j]!=-1){
                    targetMap.put(num,board[i][j]);
                }
                num++;
                if(!flag) j++;
                else j--;
            }
            flag=!flag;
        }
        Queue<Integer> q=new LinkedList<>();
        points.add(1);
        q.offer(1);
        int count=0;
        while(!q.isEmpty()){
            int size=q.size();
            count++;
            System.out.println(count+":");
            for(int k=0;k<size;k++){
                int point=q.poll();
                for(int i=1;i<=6;i++){
                    int target=(point+i)%n2;
                    // 这里是关键，如果能跳则必须跳
                    if(targetMap.containsKey(target)){
                        target=targetMap.get(target)%n2;    
                    }
                    if(!points.contains(target)){
                        points.add(target);
                        q.offer(target);
                        System.out.println(target);      
                    }
                }
            }
            if(points.contains(0)) break;
        }
        return points.contains(0)?count:-1;
    }
}
```
