## 题解

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    private Map<Integer,Integer> index_map=new HashMap<>();

    private TreeNode myBuildTree(
        int[] inorder,int[] postorder,
        int inorder_left,int inorder_right,
        int postorder_left,int postorder_right){

            if(inorder_left>inorder_right) return null;

            int root_val=postorder[postorder_right];
            int root_index_inorder=index_map.get(root_val);
            int left_tree_size=root_index_inorder-inorder_left;
            int right_tree_size=inorder_right-root_index_inorder;

            TreeNode root=new TreeNode(root_val);
            root.left=myBuildTree(inorder,postorder,
                inorder_left,root_index_inorder-1,
                postorder_left,postorder_left+left_tree_size-1);
            root.right=myBuildTree(inorder,postorder,
                root_index_inorder+1,inorder_right,
                postorder_left+left_tree_size,postorder_right-1);
            return root;

    }

    public TreeNode buildTree(int[] inorder, int[] postorder) {
        int len=inorder.length;
        for(int i=0;i<inorder.length;i++){
            index_map.put(inorder[i],i);
        }
        return myBuildTree(inorder,postorder,0,len-1,0,len-1);
    }
}
```
