```java
class Solution {
    List<List<Integer>> res=new ArrayList<>();

    void dfs(int index,int len,int[] candidates,int target,List<Integer> list){
        if(target==0){
            res.add(List.copyOf(list));
            return;
        }
        if(index==len||target<0) return;
        // 不选
        dfs(index+1,len,candidates,target,list);
        // 选
        list.add(candidates[index]);
        dfs(index,len,candidates,target-candidates[index],list);
        list.remove(list.size()-1);
    }

    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        int len=candidates.length;
        dfs(0,len,candidates,target,new ArrayList<Integer>());
        return res;
    }
}
```

```java
class Solution {
    List<List<Integer>> res=new ArrayList<>();

    void dfs(int index,int len,int[] candidates,int target,List<Integer> list){
        if(target==0){
            res.add(List.copyOf(list));
            return;
        }
        if(index==len||target<0) return;
        // 剪枝
        if(candidates[index]>target) return;
        // 不选
        dfs(index+1,len,candidates,target,list);
        // 选
        list.add(candidates[index]);
        dfs(index,len,candidates,target-candidates[index],list);
        list.remove(list.size()-1);
    }

    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        int len=candidates.length;
        Arrays.sort(candidates);
        dfs(0,len,candidates,target,new ArrayList<Integer>());
        return res;
    }
}
```
