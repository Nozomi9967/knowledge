## 题解



```java
class Solution {

    private static final int SQRT_INT_MAX = (int) Math.sqrt(Integer.MAX_VALUE);

    public int mySqrt(int x) {
        int left=0;
        int right=Math.min(x,SQRT_INT_MAX)+1;
        while(left+1<right){
            int mid=(left+right)>>>1;
            if(mid*mid<=x) left=mid;
            else right=mid;
        }
        return left;
    }
}
```

```java
class Solution {

        long l=1,r=x/2;

        if(x==1) return 1;

        while(l<=r){

            long m=(r-l)/2+l;

            if(m*m>x) r=m-1;

            else l=m+1;

        }

        return (int)r;

    }
```
