```java
class ExamTracker {
    private final List<Integer> times = new ArrayList<>();
    private final List<Long> preSum = new ArrayList<>();

    public ExamTracker() {
        preSum.add(0L); // 原理见题目 303. 区域和检索 - 数组不可变
    }

    public void record(int time, int score) {
        times.add(time);
        preSum.add(preSum.getLast() + score);
    }

    public long totalScore(int startTime, int endTime) {
        // times 没有重复元素，可以用库函数二分（有重复元素则不行）
        int left = Collections.binarySearch(times, startTime);
        if (left < 0) left = ~left;
        int right = Collections.binarySearch(times, endTime + 1);
        if (right < 0) right = ~right;
        return preSum.get(right) - preSum.get(left);
    }
}
```

```csharp
class ExamTracker {
    public Map<Integer,Long> map;
    public List<Integer> list;
    public ExamTracker() {
        map = new HashMap();
        list = new ArrayList();
        list.add(0);
        map.put(0,0L);
    }
    
    public void record(int time, int score) {
        int lastTime = list.get(list.size()-1);
        map.put(time,map.get(lastTime)+score);
        list.add(time);
    }
    
    public long totalScore(int startTime, int endTime) {
        int startIndex = getLastIndex(startTime);
        int endIndex = getLastIndex(endTime);
        if(list.get(startIndex)==startTime) startIndex-=1;
        long ans=map.get(list.get(endIndex)) - map.get(list.get(startIndex));
        return ans;
    }

    public int getLastIndex(int time){
        int l=0;
        int r=list.size()-1;
        while(l<r){
            int mid = l+r+1>>1;
            if(list.get(mid)<=time){
                l=mid;
            }else{
                r=mid-1;
            }
        }
        return l;
    }
}
```
