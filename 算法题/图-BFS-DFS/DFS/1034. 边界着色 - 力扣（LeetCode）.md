提示

给你一个大小为 `m x n` 的整数矩阵 `grid` ，表示一个网格。另给你三个整数 `row` 、 `col` 和 `color` 。网格中的每个值表示该位置处的网格块的颜色。

如果两个方块在任意 4 个方向上相邻，则称它们 **相邻** 。

如果两个方块具有相同的颜色且相邻，它们则属于同一个 **连通分量** 。

**连通分量的边界** 是指连通分量中满足下述条件之一的所有网格块：

- 在上、下、左、右任意一个方向上与不属于同一连通分量的网格块相邻
- 在网格的边界上（第一行/列或最后一行/列）

请你使用指定颜色 `color` 为所有包含网格块 `grid[row][col]` 的 **连通分量的边界** 进行着色。

并返回最终的网格 `grid` 。

**示例 1：**

```
输入：grid = [[1,1],[1,2]], row = 0, col = 0, color = 3
输出：[[3,3],[3,2]]
```

**示例 2：**

```
输入：grid = [[1,2,2],[2,3,2]], row = 0, col = 1, color = 3
输出：[[1,3,3],[2,3,3]]
```

**示例 3：**

```
输入：grid = [[1,1,1],[1,1,1],[1,1,1]], row = 1, col = 1, color = 2
输出：[[2,2,2],[2,1,2],[2,2,2]]
```

**提示：**

- `m == grid.length`
- `n == grid[i].length`
- `1 <= m, n <= 50`
- `1 <= grid[i][j], color <= 1000`
- `0 <= row < m`
- `0 <= col < n`

```java
class Solution {

    private int m;

    private int n;

    private boolean[][] v;

    private int cur;

    // 用一个列表来记录所有确定是边框的点，DFS 结束后统一着色

    private List<int[]> borders = new ArrayList<>();

  

    private void dfs(int[][] grid, int i, int j, int color) {

        v[i][j] = true;

        boolean isBorder = false;

  

        // 定义上下左右四个方向

        int[] di = {-1, 1, 0, 0};

        int[] dj = {0, 0, -1, 1};

  

        for (int k = 0; k < 4; k++) {

            int ni = i + di[k];

            int nj = j + dj[k];

  

            // 1. 如果越界，说明当前点 (i, j) 贴着矩阵边界，它是边框

            if (ni < 0 || nj < 0 || ni >= m || nj >= n) {

                isBorder = true;

            }

            // 2. 如果没越界，但是邻居的颜色和初始颜色不同，它也是边框

            else if (grid[ni][nj] != cur) {

                isBorder = true;

            }

            // 3. 只有当邻居合法、颜色一致、且没有被访问过时，才继续往下 DFS

            else if (!v[ni][nj]) {

                dfs(grid, ni, nj, color);

            }

        }

  

        // 如果四个方向检查完，确定它是边框，记录下来

        if (isBorder) {

            borders.add(new int[]{i, j});

        }

    }

  

    public int[][] colorBorder(int[][] grid, int row, int col, int color) {

        m = grid.length;

        n = grid[0].length;

        v = new boolean[m][n];

        cur = grid[row][col];

  

        // 1. 跑一遍 DFS 找出所有属于该连通分量的边框

        dfs(grid, row, col, color);

  

        // 2. 统一对边框进行着色

        for (int[] cell : borders) {

            grid[cell[0]][cell[1]] = color;

        }

  

        return grid;

    }

}
```
