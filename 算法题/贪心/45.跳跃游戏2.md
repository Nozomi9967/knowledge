// 我的题解
// 思路：定义一个step数组，每个元素值记录着对应nums元素的从0出发到该元素的最小跳跃值。
//  	从每个nums中的元素出发，更新它所能到达的nums元素的step值

class Solution {
    public int jump(int[] nums) {
        int len=nums.length;
        int[] step=new int[len];
        step[0]=0;
        for(int i=0;i<len;i++){
            for(int j=1;j<=nums[i];j++){
                if(i+j>=len) break;
                if(step[i+j]==0 || step[i+j]>step[i]+1) step[i+j]=step[i]+1;
            }
        }
        return step[len-1];
    }
}
