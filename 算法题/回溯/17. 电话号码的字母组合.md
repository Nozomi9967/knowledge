## 题目

https://leetcode.cn/problems/letter-combinations-of-a-phone-number/description/?envType=study-plan-v2&envId=top-interview-150





## 题解

```java
class Solution {

    	// 匿名内部类 + 实例初始化块
        List<List<Character>> mapping = new ArrayList<>() {{
        add(List.of('a', 'b', 'c'));       // 对应数字2
        add(List.of('d', 'e', 'f'));       // 对应数字3
        add(List.of('g', 'h', 'i'));       // 对应数字4
        add(List.of('j', 'k', 'l'));       // 对应数字5
        add(List.of('m', 'n', 'o'));       // 对应数字6
        add(List.of('p', 'q', 'r', 's'));  // 对应数字7
        add(List.of('t', 'u', 'v'));       // 对应数字8
        add(List.of('w', 'x', 'y', 'z'));  // 对应数字9
    }};

    void dfs(int index,int len,String digits,String s,List<String> res){
        if(index==len){
             res.add(s);
             return;
        }
        char c=digits.charAt(index);
        List<Character> list=mapping.get(c-'0'-2);
        for(int i=0;i<list.size();i++){
            dfs(index+1,len,digits,s+list.get(i),res);
        }
    }

    public List<String> letterCombinations(String digits) {
        int len=digits.length();
        List<String> res=new ArrayList<>();
        dfs(0,len,digits,new String(),res);
        return res;
    }
}
```

```java
class Solution {

  
  

    public static final char[][] map=new char[][]{

        {},{},

        {'a','b','c'},{'d','e','f'},{'g','h','i'},

        {'j','k','l'},{'m','n','o'},{'p','q','r','s'},

        {'t','u','v'},{'w','x','y','z'}

    };

  

    void dfs(char[] digits,List<String> ans,StringBuilder sb){

  

        int i=sb.length();

  

        if(i==digits.length){

            ans.add(sb.toString());return;

        }

  

        for(char c:map[digits[i]-'0']){

            sb.append(c);

            dfs(digits,ans,sb);

            sb.deleteCharAt(sb.length()-1);

        }

    }

  

    public List<String> letterCombinations(String digits) {

        List<String> ans=new ArrayList<>();

        dfs(digits.toCharArray(),ans,new StringBuilder());

        return ans;

    }

}
```
