## 题目

https://leetcode.cn/problems/n-queens-ii/description/?envType=study-plan-v2&envId=top-interview-150



## 题解

```java
class Solution {

    private int count=0;
    // 记录某列visited状态
    private boolean[] col;
    // 记录正对角线visited状态
    private boolean[] b1;
    // 记录副对角线visited状态
    private boolean[] b2;

    void dfs(int index,int n){
        if(index==n){
            count++;
            return;
        }
        for(int i=0;i<n;i++){
            // 正对角-->行列下标之和相同
            int num1=index+i;
            // 副对角-->行列下标之差相同  加n-1为了保证正数
            int num2=index-i+n-1;
            if(!col[i]&&!b1[num1]&&!b2[num2]){
                col[i]=b1[num1]=b2[num2]=true;
                dfs(index+1,n);
                // 回溯
                col[i]=b1[num1]=b2[num2]=false;
            }
        }
    }

    public int totalNQueens(int n) {
        col=new boolean[n];
        // 拿n=4举例一下，就知道2*n-1怎么来的了
        b1=new boolean[2*n-1];
        b2=new boolean[2*n-1];
        dfs(0,n);
        return count;
    }
}
```
