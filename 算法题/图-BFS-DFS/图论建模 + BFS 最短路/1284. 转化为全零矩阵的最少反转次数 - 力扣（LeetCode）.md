给你一个 `m x n` 的二进制矩阵 `mat` 。每一步，你可以选择一个单元格并将它反转（反转表示 `0` 变 `1` ， `1` 变 `0` ）。如果存在和它相邻的单元格，那么这些相邻的单元格也会被反转。相邻的两个单元格共享同一条边。

请你返回将矩阵 `mat` 转化为全零矩阵的 *最少反转次数* ，如果无法转化为全零矩阵，请返回 `-1` 。

**二进制矩阵** 的每一个格子要么是 `0` 要么是 `1` 。

**全零矩阵** 是所有格子都为 `0` 的矩阵。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2019/12/13/matrix.png)

```
输入：mat = [[0,0],[0,1]]
输出：3
解释：一个可能的解是反转 (1, 0)，然后 (0, 1) ，最后是 (1, 1) 。
```

**示例 2：**

```
输入：mat = [[0]]
输出：0
解释：给出的矩阵是全零矩阵，所以你不需要改变它。
```

**示例 3：**

```
输入：mat = [[1,0,0],[1,0,0]]
输出：-1
解释：该矩阵无法转变成全零矩阵
```

import java.util.*;

  

class Solution {

    // 五个方向：上、下、左、右、自身

    private static final int[][] DIRS = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}, {0, 0}};

  

    // 将矩阵编码为一个整数：按行优先顺序把每个格子的值拼接成二进制数

    private int encode(int[][] mat, int m, int n) {

        int x = 0;

        for (int i = 0; i < m; i++) {

            for (int j = 0; j < n; j++) {

                x = x * 2 + mat[i][j];

            }

        }

        return x;

    }

  

    // 将整数解码回 m x n 矩阵，与encode的顺序相反

    private int[][] decode(int x, int m, int n) {

        int[][] mat = new int[m][n];

        for (int i = m - 1; i >= 0; i--) {

            for (int j = n - 1; j >= 0; j--) {

                mat[i][j] = x & 1;

                x >>= 1;

            }

        }

        return mat;

    }

  

    // 翻转以(i, j)为中心的十字形位置（自身及上下左右邻居）

    private void convert(int[][] mat, int m, int n, int i, int j) {

        for (int[] d : DIRS) {

            int i0 = i + d[0], j0 = j + d[1];

            if (i0 >= 0 && i0 < m && j0 >= 0 && j0 < n) {

                mat[i0][j0] ^= 1;

            }

        }

    }

  

    public int minFlips(int[][] mat) {

        int m = mat.length, n = mat[0].length;

        int xStart = encode(mat, m, n);

        int step = 0;

        if (xStart == 0) {

            return step;

        }

  

        Set<Integer> visited = new HashSet<>();

        Queue<Integer> queue = new LinkedList<>();

        visited.add(xStart);

        queue.offer(xStart);

  

        while (!queue.isEmpty()) {

            step++;

            int size = queue.size();

            for (int t = 0; t < size; t++) {

                int[][] status = decode(queue.poll(), m, n);

                for (int i = 0; i < m; i++) {

                    for (int j = 0; j < n; j++) {

                        convert(status, m, n, i, j);

                        int xCur = encode(status, m, n);

                        if (xCur == 0) {

                            return step;

                        }

                        if (!visited.contains(xCur)) {

                            visited.add(xCur);

                            queue.offer(xCur);

                        }

                        // 翻转两次相当于恢复原状，用于还原status以便尝试下一个位置

                        convert(status, m, n, i, j);

                    }

                }

            }

        }

  

        return -1;

    }

}
