## 题目

https://leetcode.cn/problems/surrounded-regions/description/?envType=study-plan-v2&envId=top-interview-150





## 题解

```java
class Solution {

    void dfs(char[][] board,int r,int c){
        int nr=board.length;
        int nc=board[0].length;
        if(r<0||c<0||r>=nr||c>=nc||board[r][c]=='X'||board[r][c]=='@') return;
        System.out.println("@"+r+c);
        board[r][c]='@';
        dfs(board,r-1,c);
        dfs(board,r+1,c);
        dfs(board,r,c-1);
        dfs(board,r,c+1);
        return;
    }

    public void solve(char[][] board) {
        int nr=board.length;
        int nc=board[0].length;
        for(int i=0;i<nr;i++){
            for(int j=0;j<nc;j++){
                if(board[i][j]=='O'&&(i==0||j==0||i==nr-1||j==nc-1)) dfs(board,i,j);
            }
        }
        for(int i=0;i<nr;i++){
            for(int j=0;j<nc;j++){
                if(board[i][j]!='@') board[i][j]='X';
                else board[i][j]='O';
            }
        }
    }
}
```
