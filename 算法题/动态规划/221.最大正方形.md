## 题解



```java
class Solution {
    public int maximalSquare(char[][] matrix) {
        int m=matrix.length;
        int n=matrix[0].length;
        int[][] dp=new int[m][n];
        int ans=Integer.MIN_VALUE;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(matrix[i][j]=='1'){
                    dp[i][j]=1;
                    // 加厚正方形
                    if(i-1>=0&&j-1>=0&&dp[i-1][j-1]>=1){
                        int len=dp[i-1][j-1];
                        int a=i,b=j;
                        while(len!=0){
                            if(matrix[--a][j]!='1'||matrix[i][--b]!='1') break;
                            len--;
                        }
                        if(len==0) dp[i][j]=dp[i-1][j-1]+1;
                        // 这个情况比较特殊
                        else dp[i][j]=dp[i-1][j-1]-len+1;
                    }
                }
                ans=Math.max(ans,dp[i][j]);
            }
        }
        return ans*ans;
    }
}
```
