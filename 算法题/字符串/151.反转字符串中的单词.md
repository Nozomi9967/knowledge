```java
// 我的
// 时间O(n) 空间O(1)
// 需要用到StringBuilder的Api
// 从后往前遍历s，每次遇到单词的第一个字符尾插到结果字符串
// 单词的其它字符插到结果字符串中 (结果字符串长度-当前单词长度) 索引的位置
// 每次插入新单词时，先插入一个空格，如果当前没有单词则不插
class Solution {
    public String reverseWords(String s) {
        int len=s.length();
        boolean isHead=false;
        StringBuilder reverse=new StringBuilder("");
        int curWordLen=0;
        for(int i=len-1;i>=0;i--){
            if(s.charAt(i)==' '){
                isHead=false;
                curWordLen=0;
                 continue;
            }
            if(!isHead){
                if(reverse.length()!=0) reverse.append(' ');
                reverse.append(s.charAt(i));
                isHead=!isHead;
            }else{
                reverse.insert(reverse.length()-curWordLen,s.charAt(i));
            }
            curWordLen++;
            // System.out.println(reverse.toString());
        }
        return reverse.toString();
    }
}
```
