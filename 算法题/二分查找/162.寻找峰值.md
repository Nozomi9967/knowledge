## 题解

```java
class Solution {
    public int findPeakElement(int[] nums) {
        int left=-1,right=nums.length-1;
        while(left+1<right){
            int middle=left+(right-left)/2;
            if(nums[middle]<nums[middle+1]){
                left=middle;
            }else{
                right=middle;
            }
        }
        return right;
    }
}
```

```java
class Solution {

    public int findPeakElement(int[] nums) {

        int n=nums.length;

        if(n==1) return 0;

        int left=0,right=n-1;

        while(left<=right){

            int middle=left+(right-left)/2;

            if(middle+1>=n || nums[middle]<nums[middle+1]){

                left=middle+1;

            }else{

                right=middle-1;

            }

        }

        return left==n?n-1:left;

    }

}
```
