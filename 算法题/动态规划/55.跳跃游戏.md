// 我的题解
class Solution {
    public boolean canJump(int[] nums) {
        int len=nums.length;
        if(len==1) return true;
        boolean[] hash=new boolean[len];
            for(int j=len-1;j>=0;j--){
                if(nums[j]+j>=len-1){
                    hash[j]=true;
                }else{
                    for(int k=nums[j];k>0;k--){
                        if(hash[k+j]){
                            hash[j]=true;
                            break;
                        }
                    }
                }
            }
        return hash[0];
    }
}

// 官方的
public class Solution {
    public boolean canJump(int[] nums) {
        int n = nums.length;
        int rightmost = 0;
        for (int i = 0; i < n; ++i) {
            if (i <= rightmost) {
                rightmost = Math.max(rightmost, i + nums[i]);
                if (rightmost >= n - 1) {
                    return true;
                }
            }
        }
        return false;
    }
}
