## 题解

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<List<Integer>> zigzagLevelOrder(TreeNode root) {
                List<List<Integer>> res=new ArrayList<>();
        if(root==null) return res;
        List<TreeNode> floor=List.of(root);
        boolean flag=false;
        res.add(List.of(root.val));
        while(!floor.isEmpty()){
            flag=!flag;
            List<TreeNode> ceil=floor;
            floor=new ArrayList<>();
            List<Integer> list=new ArrayList<>();
            for(int i=0;i<ceil.size();i++){
                TreeNode n=ceil.get(i);
                if(n.left!=null){
                    floor.add(n.left);
                    list.add(n.left.val);
                }
                if(n.right!=null){
                    floor.add(n.right);
                    list.add(n.right.val);
                }
            }
            if(list.size()!=0){
                if(flag) Collections.reverse(list);
                res.add(list);
            }
        }
        return res;
    }
}
```
